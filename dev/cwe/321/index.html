    Use of Hard-coded Cryptographic Key

    항목 내용
    추상화 수준 변형(Variant)
    상태 초안(Draft)

    가이드 적용성 조건부 — 암호 기능 사용
    적용 플랫폼 언어: 언어 무관, Go · 기술: 산업제어시스템·운영기술(ICS/OT)
    원문 MITRE CWE-321 (CWE 5.0)

    한 줄 요약

    하드코드되어 바꿀 수 없는 암호화 키를 사용해 생기는 약점이다.

    → 나쁜 예·좋은 예

    설명(Description)¶

    제품이 하드코드되어 바꿀 수 없는 암호화 키를 사용하는 약점이다.

    원문(Description)

    The product uses a hard-coded, unchangeable cryptographic key.

    발생 단계(Modes Of Introduction)¶

    CWE 단계(Phase) 수명주기 단계(가이드 대응) 설명
    아키텍처·설계(Architecture and Design) 설계 실현(REALIZATION): 이 약점은 아키텍처 보안 전술을 구현하는 과정에서 생긴다.

    영향(Common Consequences)¶

    영역 영향 설명
    접근 통제 보호 메커니즘 우회 / 권한 획득 또는 신원 도용 / 애플리케이션 데이터 읽기 하드코드된 암호화 키를 쓰면 악의적 사용자가 해당 계정으로 접근할 것이 거의 확실하며, 암호화된 데이터가 복구될 가능성이 크게 높아진다.

    완화책(Potential Mitigations)¶

    원문 완화책 1건(수명주기 단계로 묶은 규칙은 색인에 있다).

    요구사항 분석 (본문 제3장)¶

    원문 완화책 없음

    설계 (본문 제4장)¶

    CWE 단계(Phase) 전략 요지
    아키텍처·설계(Architecture and Design) — 예방 방법은 하드코드된 비밀번호 저장의 경우와 같다.
    원문

    Prevention schemes mirror that of hard-coded password storage.

    구현 (본문 제5장)¶

    원문 완화책 없음

    통합 및 시험 (본문 제6장)¶

    원문 완화책 없음. 이 단계의 기준선 활동: T-08 운용 빌드 산출물 검사

    유지보수 (본문 제7장)¶

    원문 완화책 없음. 이 단계의 기준선 활동: M-06 운용 보호

    예시(나쁜 예·좋은 예)¶

    아래 코드는 이 가이드가 다음 상황을 가정해 새로 작성한 것이다. 장비가 데이터링크 메시지·정비 명령·정비 기록을 보호하는 키를 소스코드 상수로 넣어 펌웨어·배포본과 함께 내보낸다. 같은 배포본을 쓰는 모든 장비가 같은 키를 갖게 되고, 이미지 하나만 빼내면 strings 같은 도구로 키를 찾을 수 있다. C·C++·Java 예시는 자기 실행 파일(클래스 파일)에서 키 바이트를 찾아 보고, C# 예시는 분석기 경고(CA5390)로, Python 예시는 컴파일한 .pyc에서 키의 16진 문자열을 찾아 확인한다. 언어마다 분석기·새니타이저·실행 결과로 나쁜 예는 잡히고 좋은 예는 잡히지 않음을 확인했다. 좋은 예는 그 약점 하나를 막는 최소 예이며 그대로 운용 코드가 되는 것은 아니다. 언어 상자를 눌러 펼친다.

    C

    나쁜 예

    #include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* 모든 장비의 펌웨어에 같은 링크 키가 상수로 들어간다 */
static const unsigned char LINK_KEY[] = "EXAMPLE-ONLY-KEY";

/* 검증된 암호 모듈에 키를 넘기는 자리(시험용 대체: 복사만 한다) */
static unsigned char active_key[16];
static void crypto_set_link_key(const unsigned char *k) { memcpy(active_key, k, sizeof active_key); }

/* 이미지를 빼낸 사람이 하는 일: 실행 파일에서 사용 중인 키 바이트를 찾는다 */
static long find_in_image(const char *image, const unsigned char *pat, size_t n)
{
    FILE *fp = fopen(image, "rb");
    if (fp == NULL) return -2;
    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    rewind(fp);
    unsigned char *buf = size > 0 ? malloc((size_t)size) : NULL;
    long at = -1;
    if (buf != NULL && fread(buf, 1, (size_t)size, fp) == (size_t)size) {
        for (long i = 0; at < 0 && i + (long)n <= size; ++i) {
            if (memcmp(buf + i, pat, n) == 0) at = i;
        }
    }
    free(buf);
    fclose(fp);
    return at;
}

int main(int argc, char **argv)
{
    (void)argc;
    crypto_set_link_key(LINK_KEY);
    long at = find_in_image(argv[0], active_key, sizeof active_key);
    if (at >= 0) printf("KEY FOUND IN IMAGE at offset %ld\n", at);
    else puts(at == -2 ? "image not readable" : "key not present in image");
    return 0;
}

    링크 키를 static const 배열로 넣어 실행 이미지의 읽기 전용 영역에 그대로 남는다. 장비 한 대의 펌웨어만 덤프해도 같은 이미지를 쓰는 전체 장비의 키가 드러나고, 키를 바꾸려면 재빌드·재배포해야 한다.

    좋은 예

    #define _CRT_RAND_S
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* 키 저장소 인터페이스: 장비마다 생산 단계에서 주입한 고유 키를 보안요소·eFuse에서 읽는다.
   시험용 대체 구현: 운영체제 난수로 채운다(실제 장비에서는 보안요소 드라이버로 바꾼다). */
static int keystore_read_link_key(unsigned char *out, size_t n)
{
    for (size_t i = 0; i < n; ++i) {
        unsigned int r;
        if (rand_s(&r) != 0) return -1;
        out[i] = (unsigned char)r;
    }
    return 0;
}

static unsigned char active_key[16];

static long find_in_image(const char *image, const unsigned char *pat, size_t n)
{
    FILE *fp = fopen(image, "rb");
    if (fp == NULL) return -2;
    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    rewind(fp);
    unsigned char *buf = size > 0 ? malloc((size_t)size) : NULL;
    long at = -1;
    if (buf != NULL && fread(buf, 1, (size_t)size, fp) == (size_t)size) {
        for (long i = 0; at < 0 && i + (long)n <= size; ++i) {
            if (memcmp(buf + i, pat, n) == 0) at = i;
        }
    }
    free(buf);
    fclose(fp);
    return at;
}

int main(int argc, char **argv)
{
    (void)argc;
    if (keystore_read_link_key(active_key, sizeof active_key) != 0) {
        puts("link key not provisioned: link disabled");   /* 기본 키로 대신하지 않는다 */
        return 1;
    }
    long at = find_in_image(argv[0], active_key, sizeof active_key);
    if (at >= 0) printf("KEY FOUND IN IMAGE at offset %ld\n", at);
    else puts(at == -2 ? "image not readable" : "key not present in image");
    return 0;
}

    키는 생산 단계에서 장비마다 주입하고 보안요소·eFuse에서 읽는 키 저장소 인터페이스로 받는다(예시의 rand_s 구현은 시험용 대체). 키가 없으면 기본 키로 대신하지 않고 링크를 열지 않는다(SOC-04·D-05).

    검증: 실행 결과 — 나쁜 예에서 KEY FOUND IN IMAGE 검출, 좋은 예에서 미검출·key not present in image 출력 확인 · 참고: MITRE CWE-321 Demonstrative Example 1(하드코드된 암호 키로 비밀번호 확인) 참고, 이 가이드가 새로 작성 / KISA·행정안전부 「C 시큐어코딩 가이드」(3판) 보안기능 2-9 '하드코드된 암호화키'(인쇄 107쪽) 참고

    C++

    나쁜 예

    #include <algorithm>
#include <array>
#include <cstdio>
#include <fstream>
#include <iterator>
#include <vector>

// 정비 명령 인증에 쓰는 키를 소스코드에 상수로 넣었다(모든 장비가 같은 키)
constexpr std::array<unsigned char, 16> kMaintKey{'E', 'X', 'A', 'M', 'P', 'L', 'E', '-',
                                                  'O', 'N', 'L', 'Y', '-', 'K', 'E', 'Y'};

class CommandAuthenticator {   // 검증된 암호 모듈을 감싸는 자리(시험용: 키만 보관)
public:
    explicit CommandAuthenticator(const std::array<unsigned char, 16> &key) : key_(key) {}
    const std::array<unsigned char, 16> &key() const { return key_; }
private:
    std::array<unsigned char, 16> key_;
};

// 이미지를 빼낸 사람이 하는 일: 실행 파일에서 키 바이트를 찾는다
static bool image_contains(const char *image, const std::array<unsigned char, 16> &key) {
    std::ifstream in(image, std::ios::binary);
    const std::vector<unsigned char> raw((std::istreambuf_iterator<char>(in)), std::istreambuf_iterator<char>());
    return std::search(raw.begin(), raw.end(), key.begin(), key.end()) != raw.end();
}

int main(int, char **argv) {
    const CommandAuthenticator auth(kMaintKey);
    std::puts(image_contains(argv[0], auth.key()) ? "KEY FOUND IN IMAGE" : "key not present in image");
    return 0;
}

    정비 명령 인증 키를 constexpr 배열로 소스코드에 넣는다. 키가 실행 파일에 그대로 실려 배포본을 가진 사람은 누구나 키를 얻는다.

    좋은 예

    #include <windows.h>
#include <bcrypt.h>
#include <algorithm>
#include <array>
#include <cstdio>
#include <fstream>
#include <iterator>
#include <optional>
#include <vector>

using Key = std::array<unsigned char, 16>;

// 키 저장소 인터페이스: 장비별 고유 키를 생산 단계에서 주입하고 보안요소에서 읽는다.
// 시험용 대체 구현: 운영체제 난수 생성기로 채운다(실제 장비에서는 보안요소 드라이버로 바꾼다).
static std::optional<Key> keystore_maint_key() {
    Key k{};
    if (!BCRYPT_SUCCESS(BCryptGenRandom(nullptr, k.data(), static_cast<ULONG>(k.size()),
                                        BCRYPT_USE_SYSTEM_PREFERRED_RNG))) {
        return std::nullopt;
    }
    return k;
}

class CommandAuthenticator {
public:
    explicit CommandAuthenticator(const Key &key) : key_(key) {}
    ~CommandAuthenticator() { SecureZeroMemory(key_.data(), key_.size()); }   // 사용 후 소거
    const Key &key() const { return key_; }
private:
    Key key_;
};

static bool image_contains(const char *image, const Key &key) {
    std::ifstream in(image, std::ios::binary);
    const std::vector<unsigned char> raw((std::istreambuf_iterator<char>(in)), std::istreambuf_iterator<char>());
    return std::search(raw.begin(), raw.end(), key.begin(), key.end()) != raw.end();
}

int main(int, char **argv) {
    const auto key = keystore_maint_key();
    if (!key) {
        std::puts("maintenance key not provisioned: commands disabled");
        return 1;
    }
    const CommandAuthenticator auth(*key);
    std::puts(image_contains(argv[0], auth.key()) ? "KEY FOUND IN IMAGE" : "key not present in image");
    return 0;
}

    키 저장소 함수에서 장비별 키를 받고(예시의 BCryptGenRandom 구현은 시험용 대체), 키를 못 받으면 정비 명령 기능을 끈다. 키를 보관한 객체는 소멸할 때 SecureZeroMemory로 지운다(D-05). 지우는 것은 이 객체의 값이므로, 키 저장소 함수가 돌려준 사본(main의 key)도 운용 코드에서는 따로 지운다.

    검증: 실행 결과 — 나쁜 예에서 KEY FOUND IN IMAGE 검출, 좋은 예에서 미검출·key not present in image 출력 확인 · 참고: MITRE CWE-321 Demonstrative Example 1(하드코드된 암호 키로 비밀번호 확인) 참고, 이 가이드가 새로 작성

    C#

    나쁜 예

    using System.Security.Cryptography;
using System.Text;

// 링크 메시지 보호 키를 소스코드에 상수로 넣었다(모든 배포본이 같은 키)
byte[] linkKey = Encoding.ASCII.GetBytes("EXAMPLE-ONLY-KEY-EXAMPLE-ONLY-32");
byte[] order = Encoding.ASCII.GetBytes("WP7 HOLD ALT=1500");
byte[] nonce = RandomNumberGenerator.GetBytes(12);
byte[] tag = new byte[16];
byte[] sealedOrder = new byte[order.Length];

using var gcm = new AesGcm(linkKey, tag.Length);   // CA5390: 하드코드된 키
gcm.Encrypt(nonce, order, sealedOrder, tag);
Console.WriteLine($"sealed {sealedOrder.Length} bytes");

    링크 보호 키를 문자열 상수에서 만들어 AesGcm에 넘긴다. 키가 어셈블리에 그대로 남고, .NET 분석기가 CA5390(하드코드된 암호화 키)으로 경고한다.

    좋은 예

    using System.Security.Cryptography;
using System.Text;

byte[] linkKey = LinkKeyStore.Read();              // 키 저장소에서 장비별 키를 받는다
byte[] order = Encoding.ASCII.GetBytes("WP7 HOLD ALT=1500");
byte[] nonce = RandomNumberGenerator.GetBytes(12);
byte[] tag = new byte[16];
byte[] sealedOrder = new byte[order.Length];

using (var gcm = new AesGcm(linkKey, tag.Length))
{
    gcm.Encrypt(nonce, order, sealedOrder, tag);
}
CryptographicOperations.ZeroMemory(linkKey);       // 사용 후 소거
Console.WriteLine($"sealed {sealedOrder.Length} bytes");

static class LinkKeyStore
{
    // 운용 장비에서는 HSM·보안요소·OS 키 저장소에서 읽는다.
    // 시험용 대체 구현: 실행 중에 난수로 만든다(코드·배포본에는 키가 없다).

    나쁜 예

    import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;

public class Example {
    // 정비 기록 암호화 키를 소스코드 상수로 넣었다(배포본마다 같은 키)
    private static final String RECORD_KEY = "EXAMPLE-ONLY-KEY";

    static SecretKey recordKey() {
        return new SecretKeySpec(RECORD_KEY.getBytes(StandardCharsets.US_ASCII), "AES");
    }

    // 배포본을 손에 넣은 사람이 하는 일: 클래스 파일에서 키 바이트를 찾는다
    static boolean classFileContains(byte[] key) throws IOException {
        try (InputStream in = Example.class.getResourceAsStream("Example.class")) {
            if (in == null) {
                return false;
            }
            String raw = new String(in.readAllBytes(), StandardCharsets.ISO_8859_1);
            return raw.contains(new String(key, StandardCharsets.ISO_8859_1));
        }
    }

    public static void main(String[] args) throws IOException {
        SecretKey key = recordKey();
        System.out.println(classFileContains(key.getEncoded()) ? "KEY FOUND IN IMAGE" : "key not present in image");
    }
}

    정비 기록 암호화 키를 문자열 상수로 둔다. 상수는 클래스 파일의 상수 풀에 그대로 남아 javap나 압축 해제만으로 읽힌다.

    import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.security.GeneralSecurityException;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;

public class Example {
    // 키 저장소 인터페이스: 운용 장비에서는 KeyStore(PKCS#11 HSM 등)에서 별칭으로 키를 꺼낸다.
    interface KeyProvider {
        SecretKey recordKey() throws GeneralSecurityException;
    }

    // 시험용 대체 구현: 실행 중에 키를 만든다(코드·배포본에는 키가 없다)
    static final KeyProvider TEST_PROVIDER = () -> {
        KeyGenerator kg = KeyGenerator.getInstance("AES");
        kg.init(256);
        return kg.generateKey();
    };

    static boolean classFileContains(byte[] key) throws IOException {
        try (InputStream in = Example.class.getResourceAsStream("Example.class")) {
            if (in == null) {
                return false;
            }
            String raw = new String(in.readAllBytes(), StandardCharsets.ISO_8859_1);
            return raw.contains(new String(key, StandardCharsets.ISO_8859_1));
        }
    }

    public static void main(String[] args) throws IOException, GeneralSecurityException {
        SecretKey key = TEST_PROVIDER.recordKey();
        System.out.println(classFileContains(key.getEncoded()) ? "KEY FOUND IN IMAGE" : "key not present in image");
    }
}

    좋은 예

    import hashlib
import hmac
import os
import py_compile

KEY_PATH = "maint_log.key"           # 예시 값. 실제로는 설치 설정에 고정한 절대 경로를 쓴다
KEY_LEN = 32


def load_key():
    # 키는 설치 때 장비마다 넣는 키 파일에서 읽는다. 소스·배포본에는 없다
    try:
        with open(KEY_PATH, "rb") as f:
            key = f.read()
    except FileNotFoundError:
        return None
    return key if len(key) == KEY_LEN else None      # 길이가 틀린 키는 없는 것으로 본다


def seal(line, key):
    return line + " mac=" + hmac.new(key, line.encode(), hashlib.sha256).hexdigest()


# 시험 준비: 설치 단계가 장비별 키를 소유자만 읽을 수 있게(0o600, POSIX 기준) 넣은 상태
with os.fdopen(os.open(KEY_PATH, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600), "wb") as f:
    f.write(os.urandom(KEY_LEN))
key = load_key()
print(seal("2026-10-12 HYD-2 seal replaced", key) if key else "key not provisioned: logging disabled")
image = open(py_compile.compile(__file__, cfile="tool.pyc"), "rb").read()
print("KEY FOUND IN IMAGE" if key and key in image else "no key in image")

    키는 설치 단계에서 장비마다 넣는 키 파일(또는 운영체제 키 저장소·보안요소)에서 읽고, 소스에는 키를 찾는 위치만 둔다(D-05). 키 파일은 소유자만 읽게 만들고, 경로는 절대 경로로 고정하며, 길이가 틀린 키는 쓰지 않는다. 키가 없으면 기본 키로 대신하지 않고 기능을 멈춘다. 배포 전 산출물에서 키·비밀값 문자열을 찾는 검사를 한다(T-08·LPY-12).
